    const dataPengguna = JSON.parse(localStorage.getItem('penggunaLMJ') || JSON.stringify(DEFAULT_USERS));
    const user = dataPengguna[nama];
    
    if(!user) return alert('❌ Pengguna tidak ditemukan!');
    if(user.sandi !== sandi) return alert('❌ Kata sandi salah!');
    if(user.peran !== peran) return alert('⚠️ Peran tidak sesuai! Pilih peran yang benar.');
    
    penggunaSaatIni = user;
    document.getElementById('loginLayer').style.display = 'none';
    document.getElementById('aplikasiUtama').style.display = 'block';
    document.getElementById('namaUser').textContent = `${user.peran === 'admin' ? '👑' : '🧑‍💼'} ${user.nama}`;
    
    if(user.peran === 'admin') document.body.classList.add('admin');
    
    inisialisasiAplikasi();
}

// 📱 SCANNER BARCODE
function toggleScanner() {
    if(!scannerAktif) {
        document.getElementById('reader').style.display = 'block';
        document.getElementById('btnScan').textContent = '📷 Tutup Scanner';
        scannerAktif = true;
        
        html5QrCode = new Html5Qrcode("reader");
        html5QrCode.start(
            { facingMode: "environment" },
            { fps: 10, qrbox: {width:250, height:150} },
            (decodedText) => {
                document.getElementById('statusScan').className = 'status-scan berhasil';
                document.getElementById('statusScan').textContent = '✅ Terbaca: ' + decodedText;
                cariProdukBarcode(decodedText);
                html5QrCode.stop();
                document.getElementById('reader').style.display = 'none';
                document.getElementById('btnScan').textContent = '📷 Scan Barcode';
                scannerAktif = false;
            },
            (err) => {}
        ).catch(err => {
            alert('⚠️ Izinkan akses kamera di pengaturan HP!');
            scannerAktif = false;
            document.getElementById('btnScan').textContent = '📷 Scan Barcode';
        });
    } else {
        if(html5QrCode) html5QrCode.stop().catch(e=>{});
        document.getElementById('reader').style.display = 'none';
        document.getElementById('btnScan').textContent = '📷 Scan Barcode';
        document.getElementById('statusScan').className = 'status-scan';
        scannerAktif = false;
    }
}

function cariBarcode() {
    const kode = document.getElementById('inputBarcode').value.trim();
    if(!kode) return;
    cariProdukBarcode(kode);
    document.getElementById('inputBarcode').value = '';
}

function cariProdukBarcode(kodeBarcode) {
    const p = produkStok.find(x => x.barcode === kodeBarcode);
    if(!p) {
        document.getElementById('statusScan').className = 'status-scan gagal';
        document.getElementById('statusScan').textContent = '❌ Barcode tidak ditemukan: ' + kodeBarcode;
        return;
    }
    document.getElementById('statusScan').className = 'status-scan berhasil';
    document.getElementById('statusScan').textContent = '✅ ' + p.nama + ' — Rp ' + p.harga.toLocaleString('id-ID');
    tambahKeKeranjang(p.id);
    setTimeout(() => { document.getElementById('statusScan').className = 'status-scan'; }, 3000);
}

// 📦 TAMPILKAN PRODUK
function tampilProduk() {
    const grid = document.getElementById('produkGrid');
    grid.innerHTML = '';
    produkStok.forEach(p => {

        grid.innerHTML += `
            <div class="produk" onclick="${stokHabis ? 'alert(\'⚠️ Stok Habis!\')' : `tambahKeKeranjang(${p.id})`}" style="${stokHabis ? 'opacity:0.5; background:#fee2e2;' : ''}">

                <div class="produk-harga">Rp ${p.harga.toLocaleString('id-ID')}</div>

            </div>
        `;
    });
}

// ➕ TAMBAH KE KERANJANG
function tambahKeKeranjang(id) {
    const p = produkStok.find(x => x.id === id);
    const ada = keranjang.find(x => x.id === id);
    if(ada) {
        if(ada.qty >= p.stok) return alert(`⚠️ Stok tidak cukup! Tersisa: ${p.stok}`);
        ada.qty++;
    } else {
        keranjang.push({ ...p, qty:1 });
    }
    updateKeranjang();
}

function ubahQty(idx, delta) {
    const p = produkStok.find(x => x.id === keranjang[idx].id);
    const qtyBaru = keranjang[idx].qty + delta;
    if(delta > 0 && qtyBaru > p.stok) return alert(`⚠️ Stok tidak cukup! Tersisa: ${p.stok}`);
    keranjang[idx].qty = qtyBaru;
    if(keranjang[idx].qty <= 0) keranjang.splice(idx, 1);
    updateKeranjang();
}

function updateKeranjang() {
    const el = document.getElementById('daftarKeranjang');
    if(keranjang.length === 0) {
        el.innerHTML = '<p style="color:#888; text-align:center; padding:15px;">Scan barcode atau klik barang</p>';
    } else {
        el.innerHTML = '';
        keranjang.forEach((item, idx) => {
            el.innerHTML += `
                <div class="item">
                    <div class="nama">${item.nama}</div>
                    <div class="qty">
                        <button onclick="ubahQty(${idx}, -1)">−</button>
                        <span>${item.qty}</span>
                        <button onclick="ubahQty(${idx}, 1)">+</button>
                    </div>
                    <div class="harga">Rp ${(item.harga * item.qty).toLocaleString('id-ID')}</div>
                </div>
            `;
        });
    }
    hitungTotal();
}

function hitungTotal() {
    const subtotal = keranjang.reduce((sum, i) => sum + (i.harga * i.qty), 0);
    const diskonPct = parseInt(document.getElementById('diskon').value) || 0;
    const diskon = subtotal * diskonPct / 100;
    const total = subtotal - diskon;
    document.getElementById('subtotal').textContent = `Rp ${subtotal.toLocaleString('id-ID')}`;
    document.getElementById('totalAkhir').textContent = `Rp ${total.toLocaleString('id-ID')}`;
    hitungKembalian();
}

document.getElementById('diskon').addEventListener('input', hitungTotal);

function hitungKembalian() {
    const subtotal = keranjang.reduce((sum, i) => sum + (i.harga * i.qty), 0);
    const diskonPct = parseInt(document.getElementById('diskon').value) || 0;
    const total = subtotal - (subtotal * diskonPct / 100);
    const bayar = parseInt(document.getElementById('uangBayar').value) || 0;
    const kembali = bayar - total;
    document.getElementById('kembalian').textContent = `Rp ${kembali.toLocaleString('id-ID')}`;
}

// ✅ PROSES BAYAR & NOTA
function prosesBayar() {
    const subtotal = keranjang.reduce((sum, i) => sum + (i.harga * i.qty), 0);
    const diskonPct = parseInt(document.getElementById('diskon').value) || 0;
    const diskon = subtotal * diskonPct / 100;
    const total = subtotal - diskon;
    const bayar = parseInt(document.getElementById('uangBayar').value) || 0;
    const kembali = bayar - total;
    
    if(keranjang.length === 0) return alert('⚠️ Keranjang masih kosong!');
    if(bayar < total) return alert(`⚠️ Uang kurang! Kurang Rp ${(total - bayar).toLocaleString('id-ID')}`);
    
    // Kurangi stok
    keranjang.forEach(item => {
        const p = produkStok.find(x => x.id === item.id);
        if(p) p.stok -= item.qty;
    });
    localStorage.setItem('stokProdukLMJ', JSON.stringify(produkStok));
    tampilProduk();
    
    // Isi Nota
    const tglSekarang = new Date().toLocaleString('id-ID');
    const noTrx = `TRX-${String(noTransaksi).padStart(6, '0')}`;
    
    document.getElementById('notaNo').textContent = noTrx;
    document.getElementById('notaKasir').textContent = penggunaSaatIni.nama;
    document.getElementById('notaTanggal').textContent = tglSekarang;
    document.getElementById('notaSubtotal').textContent = `Rp ${subtotal.toLocaleString('id-ID')}`;
    document.getElementById('notaDiskon').textContent = `Rp ${diskon.toLocaleString('id-ID')} (${diskonPct}%)`;
    document.getElementById('notaTotal').textContent = `Rp ${total.toLocaleString('id-ID')}`;
    document.getElementById('notaBayar').textContent = `Rp ${bayar.toLocaleString('id-ID')}`;
    document.getElementById('notaKembali').textContent = `Rp ${kembali.toLocaleString('id-ID')}`;
    document.getElementById('notaCetakWaktu').textContent = new Date().toLocaleString('id-ID');
    
    let itemHtml = '';
    keranjang.forEach(i => {
        itemHtml += `<div class="nota-row"><span>${i.nama} x${i.qty}</span><span>Rp ${(i.harga*i.qty).toLocaleString('id-ID')}</span></div>`;
    });
    document.getElementById('notaItem').innerHTML = itemHtml;
    
    // Simpan ke Riwayat
    riwayat.unshift({
        no: noTrx, tanggal: tglSekarang, kasir: penggunaSaatIni.nama,
        subtotal, diskon, total, bayar, kembali, items: [...keranjang]
    });
    localStorage.setItem('riwayatKasirLMJ', JSON.stringify(riwayat));
    localStorage.setItem('noTransaksiLMJ', ++noTransaksi);
    
    document.getElementById('notaModal').classList.add('aktif');
}
